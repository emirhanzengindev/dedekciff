311: export function calculateCaseStrength(caseData: CaseData, session: SessionState) {
312:   const evidence = getAccessibleEvidence(caseData, session);
313:   const reliabilityScore = evidence.reduce((sum, item) => sum + item.reliability * 7, 0);
314:   const independentBonus = evidence.filter((item) => item.independentVerification).length * 12;
315:   const breadthBonus = Math.min(20, new Set(evidence.map((item) => item.type)).size * 4);
316:   const rawScore = Math.min(100, Math.round(reliabilityScore + independentBonus + breadthBonus));
317:   const weakAreas = [
318:     evidence.some((item) => item.independentVerification) ? undefined : "Bağımsız doğrulama zayıf.",
319:     session.unlockedEvidenceIds.includes("E10") ? undefined : "Ödeme iddiasının sınırları henüz çapraz kontrol edilmedi."
320:   ].filter((item): item is string => Boolean(item));
321:   return { score: rawScore, unlockedEvidenceCount: evidence.length, weakAreas };
322: }

324: export function evaluateProsecutionSubmission(
325:   caseData: CaseData,
326:   session: SessionState,
327:   submission: ProsecutionSubmission
328: ): ProsecutionDecision {
329:   const now = new Date().toISOString();
330:   const unlocked = new Set(session.unlockedEvidenceIds);
331:   const submittedEvidence = caseData.evidence.filter(
332:     (evidence) => submission.evidenceIds.includes(evidence.id) && unlocked.has(evidence.id)
333:   );
334:   const inaccessibleIds = submission.evidenceIds.filter((id) => !unlocked.has(id));
335:   const reasons: string[] = [];
336:   const componentScores: Record<string, number> = { S1: 0, S2: 0, S3: 0, S4: 0, S5: 0 };
337: 
338:   if (submission.suspectId === canonicalSuspectId && submittedEvidence.some((evidence) => evidence.id === "E04" || evidence.id === "E09")) {
339:     componentScores.S1 = 20;
340:   }
341:   if (submittedEvidence.some((evidence) => evidence.id === "E02" || evidence.id === "E03")) {
342:     componentScores.S2 = submittedEvidence.some((evidence) => evidence.id === "E03") ? 20 : 12;
343:   }
344:   if (submittedEvidence.some((evidence) => evidence.id === "E05")) {
345:     componentScores.S3 = submission.gaps || submission.contradictionAnalysis ? 16 : 8;
346:   }
347:   if (submittedEvidence.some((evidence) => evidence.independentVerification)) {
348:     componentScores.S4 = 20;
349:   } else if (submittedEvidence.length >= 3 && submission.gaps) {
350:     componentScores.S4 = 10;
351:   }
352:   if (submission.contradictionAnalysis || submission.gaps) {
353:     componentScores.S5 = 18;
354:   }
355: 
356:   let score = Object.values(componentScores).reduce((sum, value) => sum + value, 0);
357:   if (submission.suspectId !== canonicalSuspectId) {
358:     reasons.push("Yanlış şüpheli seçildi.");
359:     score = Math.min(score, 30);
360:   }
361:   if (inaccessibleIds.length > 0) {
362:     reasons.push(`Erişilmemiş veya uydurma delil sunuldu: ${inaccessibleIds.join(", ")}.`);
363:     score = Math.min(score, 30);
364:   }
365:   if (submittedEvidence.length < 3) {
366:     reasons.push("En az üç erişilmiş dayanak delil sunulmadı.");
367:     score = Math.min(score, 44);
368:   }
369:   if (!submission.gaps && !submission.contradictionAnalysis) {
370:     reasons.push("Çelişki veya eksik doğrulama analizi açıklanmadı.");
371:   }
372: 
373:   const outcome =
374:     score >= 75 && reasons.length === 0
375:       ? "accepted_for_further_action"
376:       : score >= 45 && submission.suspectId === canonicalSuspectId && inaccessibleIds.length === 0
377:         ? "returned_for_more_evidence"
378:         : "rejected";
379: 
380:   if (outcome === "accepted_for_further_action") {
381:     reasons.push("Dosya ileri incelemeye alınır; bu sonuç suçluluğun kesinleşmesi değildir.");
382:   } else if (outcome === "returned_for_more_evidence") {
383:     reasons.push("Dosya makul bağlantılar içeriyor ancak ek araştırma gerektiriyor.");
384:   }
385: 
386:   return {
387:     id: crypto.randomUUID(),
388:     outcome,
389:     score,
390:     reasons,
391:     componentScores,
392:     createdAt: now
393:   };
394: }

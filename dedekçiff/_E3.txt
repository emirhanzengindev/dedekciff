425: function addActionMarkers(session: SessionState, action: PlayerAction): SessionState {
426:   const actions = new Set(session.actions);
427:   let completedInterviews = session.completedInterviews;
428:   let interrogations = session.interrogations;
429:   let questionHistory = session.questionHistory;
430:   switch (action.type) {
431:     case "load_case":
432:       actions.add("load_case");
433:       break;
434:     case "visit_location":
435:       actions.add(`visited:${action.locationId}`);
436:       break;
437:     case "start_interview":
438:       actions.add(`interview_started:${action.characterId}`);
439:       if (!interrogations.some((interrogation) => interrogation.characterId === action.characterId)) {
440:         interrogations = [...interrogations, { characterId: action.characterId, startedAt: new Date().toISOString(), questionTopics: [] }];
441:       }
442:       break;
443:     case "complete_interview":
444:       actions.add(`interview:${action.characterId}:${action.topic ?? ""}`);
445:       completedInterviews = appendInterview(completedInterviews, action.characterId, action.topic);
446:       break;
447:     case "ask_character":
448:       actions.add(`asked:${action.characterId}:${action.topic ?? ""}`);
449:       if (action.topic) {
450:         actions.add(`topic:${action.topic}`);
451:         completedInterviews = appendInterview(completedInterviews, action.characterId, action.topic);
452:         if (action.characterId === "C02" && suspectInterrogationTopics.has(action.topic)) {
453:           interrogations = interrogations.map((interrogation) => interrogation.characterId === "C02"
454:             ? { ...interrogation, questionTopics: [...new Set([...interrogation.questionTopics, action.topic as string])] }
455:             : interrogation);
456:         }
457:       }
458:       questionHistory = [...questionHistory, {
459:         characterId: action.characterId,
460:         question: normalize(action.question).replace(/[?!.,]/gu, "").trim(),
461:         topic: action.topic ?? "",
462:         askedAt: new Date().toISOString()
463:       }];
464:       break;
465:     default:
466:       actions.add(action.type);
467:   }
468:   return { ...session, actions: [...actions], completedInterviews, interrogations, questionHistory };
469: }

471: function firstDeniedReason(caseData: CaseData, session: SessionState, action: PlayerAction): string | undefined {
472:   if (action.type === "visit_location" && !getVisibleLocations(caseData, session).some((location) => location.id === action.locationId)) {
473:     return "location_locked_or_unknown";
474:   }
475:   if (
476:     (action.type === "ask_character" || action.type === "complete_interview" || action.type === "start_interview") &&
477:     !getAccessibleCharacters(caseData, session).some((character) => character.id === action.characterId)
478:   ) {
479:     return "character_locked_or_unknown";
480:   }
481:   if (action.type === "ask_character" && action.characterId === "C02" && !session.interrogations.some((interrogation) => interrogation.characterId === "C02")) {
482:     return "interrogation_not_started";
483:   }
484: 
485:   const candidates = caseData.evidence.filter((evidence) => actionMatchesCondition(action, evidence.unlockCondition));
486:   for (const evidence of candidates) {
487:     const missing = (evidence.unlockCondition?.evidenceIds ?? []).filter(
488:       (evidenceId) => !session.unlockedEvidenceIds.includes(evidenceId)
489:     );
490:     if (missing.length > 0) return `missing_prerequisite_${missing.join("_")}`;
491:   }
492:   if (action.type === "interview_suspect") {
493:     const missing = ["E07", "E08"].filter((id) => !session.unlockedEvidenceIds.includes(id));
494:     if (missing.length > 0) return `missing_prerequisite_${missing.join("_")}`;
495:     const interrogation = session.interrogations.find((item) => item.characterId === "C02");
496:     if (!interrogation) return "interrogation_not_started";
497:     if (interrogation.questionTopics.filter((topic) => suspectInterrogationTopics.has(topic)).length < 2) return "interrogation_topics_incomplete";
498:   }
499:   return undefined;
500: }

  app.post("/api/sessions/:sessionId/questions", async (request, reply) => {
    const params = paramsSchema.parse(request.params);
    const session = await sessionStore.getSession(params.sessionId, request.userId ?? "");
    if (!session) return reply.code(404).send({ error: "session_not_found" });
    const body = questionSchema.parse(request.body);
    const questionTopic = classifyQuestionIntent(body.question);
    if (!questionTopic) {
      const answer = answerCharacterQuestion(caseData, session, body.characterId, body.question);
      return { ...answer, newlyUnlockedEvidenceIds: [], session: serializeSession(caseData, session) };
    }
    const trustedTopic = body.characterId === "C03" && questionTopic === "who_invited"
      ? "who_arranged_show"
      : body.characterId === "C03" && questionTopic === "suspect_presence"
        ? "who_arranged_show"
      : body.characterId === "C02" && questionTopic === "suspect_presence"
        ? "who_invited"
      : body.characterId === "C01" && questionTopic === "scene_objects"
        ? "show_details"
      : body.characterId === "C01" && questionTopic === "payment_recipient"
        ? "payment_details"
      : questionTopic.startsWith("payment_") || questionTopic === "payment_details"
        ? "payment_details"
        : questionTopic;
    let actionResult = applyPlayerAction(caseData, session, {
      type: "ask_character",
      characterId: body.characterId,
      question: body.question,
      topic: trustedTopic
    });
    if (actionResult.status === "denied") {
      return reply.code(403).send({ error: actionResult.reason ?? "question_not_available" });
    }
    if (body.characterId === "C02") {
      const normalizedQuestion = body.question.toLocaleLowerCase("tr-TR").replace(/[?!.,]/gu, "").trim();
      const repeated = actionResult.session.questionHistory.filter(
        (entry) => entry.characterId === "C02" && entry.question === normalizedQuestion
      ).length >= 2;
      const corroborated = actionResult.session.completedInterviews.some(
        (interview) => interview.characterId === "C03" && interview.topic === "who_arranged_show"
      );
      const interrogation = actionResult.session.interrogations.find((item) => item.characterId === "C02");
      const evidenceReady = ["E07", "E08"].every((id) => actionResult.session.unlockedEvidenceIds.includes(id));
      if (repeated && corroborated && interrogation && interrogation.questionTopics.length >= 2 && evidenceReady) {
        const completion = applyPlayerAction(caseData, actionResult.session, { type: "interview_suspect" });
        if (completion.status === "applied") actionResult = completion;
      }
    }
    await sessionStore.saveSession(actionResult.session);
    const answer = answerCharacterQuestion(caseData, actionResult.session, body.characterId, body.question);
    return {
      ...answer,
      newlyUnlockedEvidenceIds: actionResult.newlyUnlockedEvidenceIds,
      session: serializeSession(caseData, actionResult.session)
    };
  });
